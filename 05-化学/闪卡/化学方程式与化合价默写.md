## 制法类方程式

// 实验室制氧气（高锰酸钾） #card
?
$2KMnO_4 \xrightarrow{\Delta} K_2MnO_4 + MnO_2 + O_2\uparrow$

// 实验室制氧气（过氧化氢+二氧化锰） #card
?
$2H_2O_2 \xrightarrow{MnO_2} 2H_2O + O_2\uparrow$

// 实验室制二氧化碳 #card
?
$CaCO_3 + 2HCl = CaCl_2 + H_2O + CO_2\uparrow$

// 电解水 #card
?
$2H_2O \xrightarrow{\text{通电}} 2H_2\uparrow + O_2\uparrow$

// 工业炼铁（一氧化碳还原氧化铁） #card
?
$3CO + Fe_2O_3 \xrightarrow{\text{高温}} 2Fe + 3CO_2$

// 碳还原氧化铜 #card
?
$C + 2CuO \xrightarrow{\text{高温}} 2Cu + CO_2\uparrow$

## 性质类方程式

// 铁丝在氧气中燃烧 #card
?
$3Fe + 2O_2 \xrightarrow{\text{点燃}} Fe_3O_4$（火星四射、黑色固体）

// 硫在氧气中燃烧 #card
?
$S + O_2 \xrightarrow{\text{点燃}} SO_2$（明亮蓝紫色火焰）

// 二氧化碳使澄清石灰水变浑浊 #card
?
$CO_2 + Ca(OH)_2 = CaCO_3\downarrow + H_2O$

// 二氧化碳溶于水生成碳酸 #card
?
$CO_2 + H_2O = H_2CO_3$

// 盐酸检验碳酸根（碳酸钙与盐酸同制CO2式） #card
?
$CaCO_3 + 2HCl = CaCl_2 + H_2O + CO_2\uparrow$

// 生石灰与水反应 #card
?
$CaO + H_2O = Ca(OH)_2$（放热）

## 酸碱盐核心方程式

// 盐酸与氢氧化钠中和 #card
?
$HCl + NaOH = NaCl + H_2O$

// 氢氧化钠吸收二氧化碳 #card
?
$2NaOH + CO_2 = Na_2CO_3 + H_2O$

// 稀硫酸与铁 #card
?
$Fe + H_2SO_4 = FeSO_4 + H_2\uparrow$（浅绿色溶液）

// 铁与硫酸铜（湿法炼铜） #card
?
$Fe + CuSO_4 = FeSO_4 + Cu$（铁表面红色固体、溶液变浅绿）

// 硫酸铜与氢氧化钠 #card
?
$CuSO_4 + 2NaOH = Cu(OH)_2\downarrow + Na_2SO_4$（蓝色沉淀）

// 检验硫酸根（硝酸钡与硫酸钠） #card
?
$Ba(NO_3)_2 + Na_2SO_4 = BaSO_4\downarrow + 2NaNO_3$（不溶于稀硝酸的白色沉淀）

// 检验氯离子（硝酸银与盐酸/氯化钠） #card
?
$AgNO_3 + NaCl = AgCl\downarrow + NaNO_3$（不溶于稀硝酸的白色沉淀）

// 碳酸钠与盐酸（离子反应视角同制CO2） #card
?
$Na_2CO_3 + 2HCl = 2NaCl + H_2O + CO_2\uparrow$

// 高温煅烧石灰石 #card
?
$CaCO_3 \xrightarrow{\text{高温}} CaO + CO_2\uparrow$

## 溶解性口诀补充

// 常见盐的溶解性（钾钠铵盐硝酸盐） #card
?
钾钠铵盐硝酸盐，完全溶解不困难；盐酸盐不溶银亚汞；硫酸盐不溶钡和铅；碳酸盐只溶钾钠铵

// 三大不溶于稀硝酸的沉淀 #card
?
AgCl、BaSO₄（其余常见白色沉淀 CaCO₃、BaCO₃、Mg(OH)₂ 溶于酸）
